{{/* Override of Toha's shortcodes/img.html: resolves an internal "src" against the subpath-aware baseURL via helpers/get-local-path.html (upstream renders it raw). External (http/https) src values are left untouched. */}} {{- $src := .Get "src" -}} {{- if not (or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://")) -}} {{- $src = partial "helpers/get-local-path.html" $src -}} {{- end -}}
